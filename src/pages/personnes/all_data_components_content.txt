import React, { useState } from "react";
import { useSearchParams } from "react-router-dom";
import { ClipboardCheckIcon, GaugeIcon, InboxIcon, UserCheckIcon, UserPlusIcon } from "lucide-react";
import { convertirCandidature, creerCandidature, getCandidature, listerCandidatures, statuerCandidature } from "../../lib/api_ecole";
import { ModuleHero } from "../../components/layout/ModuleHero";
import { ResourceView } from "../../components/layout/ResourceView";
import { WithAnnee } from "../../components/layout/WithAnnee";
import { PivotTabs } from "../../components/ui/PivotTabs";
import { TabTransition } from "../../components/ui/TabTransition";
import { LoadPicker } from "../../components/ui/LoadPicker";
import { AsyncBlock } from "../../components/data/AsyncBlock";
import { ObjectView } from "../../components/data/ObjectView";
import { FormAction } from "../../components/form/FormAction";
import { RemplissageList } from "../../components/dashboard/RemplissageList";
import { candidatureFields, convertirFields, statuerFields } from "../../data/fields/personnel";
import { STATUT_CANDIDATURE } from "../../data/enums";
import { fromEnum } from "../../utils/options";
import { initials, str } from "../../utils/format";

const COLOR = "#DA3B01";

const nom = (r: Record<string, unknown>) => `${str(r.prenom_candidat)} ${str(r.nom_candidat)}`.trim();

export function Admissions() {
  const [params, setParams] = useSearchParams();
  const tab = params.get("tab") ?? "candidatures";
  const [statut, setStatut] = useState("");

  return (
    <div className="space-y-6">
      <ModuleHero icon={UserPlusIcon} color={COLOR} title="Admissions" subtitle="Candidatures et places disponibles" />

      <PivotTabs
        layoutKey="admissions"
        label="Rubriques des admissions"
        tabs={[
        { id: "candidatures", label: "Candidatures", icon: InboxIcon },
        { id: "places", label: "Places disponibles", icon: GaugeIcon }]
        }
        value={tab}
        onChange={(t) => setParams({ tab: t })} />

      <WithAnnee>
        {(anneeId) =>
        <TabTransition id={tab}>
            {tab === "candidatures" ?
          <ResourceView
            load={() => listerCandidatures(statut || undefined)}
            deps={[statut]}
            columns={[
            {
              key: "nom_candidat",
              label: "Candidat",
              render: (r) =>
              <span className="flex items-center gap-3">
                      <span
                  className="grid h-8 w-8 shrink-0 place-items-center rounded-full text-[12px] font-semibold"
                  style={{ background: `color-mix(in srgb, ${COLOR} 14%, transparent)`, color: COLOR }}>
                        {initials(nom(r))}
                      </span>
                      <span className="font-medium">{nom(r)}</span>
                    </span>
            },
            { key: "date_naissance" },
            { key: "nom_parent", label: "Parent" },
            { key: "telephone_parent", label: "Téléphone" },
            { key: "statut" }]
            }
            toolbar={<LoadPicker label="Statut" value={statut} onChange={setStatut} options={fromEnum(STATUT_CANDIDATURE)} placeholder="Tous" clearable />}
            actions={(reload) =>
            <FormAction label="Nouvelle candidature" icon={UserPlusIcon} variant="primary" fields={candidatureFields(anneeId)} modalSize="lg" onSubmit={creerCandidature} successMessage="Candidature enregistrée" onDone={reload} />
            }
            emptyTitle="Aucune candidature"
            detailTitle={nom}
            renderDetail={(r, ctx) =>
            <div className="space-y-5">
                    <div className="flex flex-wrap items-center gap-2 rounded-lg border border-[var(--em-border)] bg-[color-mix(in_srgb,var(--em-accent)_4%,transparent)] p-3">
                      <FormAction label="Changer le statut" icon={ClipboardCheckIcon} size="sm" fields={statuerFields} onSubmit={(b) => statuerCandidature(str(r.id), b)} successMessage="Statut mis à jour" onDone={ctx.reload} />
                      <FormAction
                  label="Inscrire comme élève"
                  icon={UserCheckIcon}
                  size="sm"
                  variant="primary"
                  fields={convertirFields(anneeId)}
                  onSubmit={(b) => convertirCandidature(str(r.id), b)}
                  successMessage="Élève inscrit"
                  showResult
                  onDone={ctx.reload} />
                    </div>
                    <AsyncBlock load={() => getCandidature(str(r.id))} deps={[str(r.id)]}>
                      {(d) => <ObjectView data={d} />}
                    </AsyncBlock>
                  </div>
            } /> :

          <section className="ecole_m-card p-5 md:p-6" aria-labelledby="places-title">
                <div className="mb-5 flex items-center gap-3">
                  <span
                className="flex h-9 w-9 shrink-0 items-center justify-center rounded-lg"
                style={{ background: `color-mix(in srgb, ${COLOR} 14%, transparent)`, color: COLOR }}>
                    <GaugeIcon className="h-[18px] w-[18px]" aria-hidden="true" />
                  </span>
                  <div>
                    <h2 id="places-title" className="text-[15px] font-semibold leading-tight">Places disponibles</h2>
                    <p className="ecole_m-muted text-[13px]">Effectifs inscrits par rapport à la capacité de chaque classe</p>
                  </div>
                </div>
                <RemplissageList anneeId={anneeId} />
              </section>
          }
          </TabTransition>
        }
      </WithAnnee>
    </div>);
}

import React, { useState } from "react";
import { KeyRoundIcon, LinkIcon, MailPlusIcon, TicketIcon, UserPlusIcon, UsersIcon } from "lucide-react";
import {
  creerCompteParent,
  genererCodeActivationParent,
  genererCodeEnveloppeParent,
  getCodeEnveloppeParent,
  getParent,
  lierEnfantParent,
  lierParentEleve,
  listerCodesEnveloppeParent,
  listerEnfantsParent,
  listerParents } from
"../../lib/api_ecole";
import { useDebounced, useTick } from "../../hooks/useAsync";
import { ModuleHero } from "../../components/layout/ModuleHero";
import { ResourceView } from "../../components/layout/ResourceView";
import { Section } from "../../components/layout/Section";

import { AsyncBlock } from "../../components/data/AsyncBlock";
import { MiniTable } from "../../components/data/MiniTable";
import { ObjectView } from "../../components/data/ObjectView";
import { EmptyState } from "../../components/data/States";
import { FormAction } from "../../components/form/FormAction";
import { codeActivationFields, lierEnfantFields, parentCreateFields, relationFields } from "../../data/fields/personnel";
import { F } from "../../utils/fieldBuilders";
import { fullName, initials, str } from "../../utils/format";

const COLOR = "#C239B3";

function ParentDetail({ parentId }: {parentId: string;}) {
  const [tick, bump] = useTick();

  return (
    <div className="space-y-6">
      <div className="rounded-lg border border-[var(--em-border)] p-4">
        <AsyncBlock load={() => getParent(parentId)} deps={[parentId]}>
          {(p) => <ObjectView data={p} hide={["nom", "prenom"]} />}
        </AsyncBlock>
      </div>

      <Section
        title="Enfants"
        actions={
        <>
            <FormAction label="Rattacher par code" icon={KeyRoundIcon} size="sm" fields={lierEnfantFields} onSubmit={(b) => lierEnfantParent(parentId, b)} successMessage="Enfant rattaché" showResult onDone={bump} />
            <FormAction label="Rattacher un élève" icon={LinkIcon} size="sm" fields={relationFields} onSubmit={(b) => lierParentEleve(parentId, b)} successMessage="Élève rattaché" onDone={bump} />
          </>
        }>
        <AsyncBlock load={() => listerEnfantsParent(parentId)} deps={[parentId, tick]} empty={<EmptyState compact title="Aucun enfant rattaché" />}>
          {(rows) =>
          <div className="overflow-hidden rounded-lg border border-[var(--em-border)]">
              <MiniTable rows={rows} rowKey={(r, i) => str(r.eleve_id) || String(i)} />
            </div>
          }
        </AsyncBlock>
      </Section>

      <Section
        title="Code d'activation du compte"
        description="Code à remettre sous enveloppe au parent pour activer son accès."
        actions={
        <FormAction
          label="Générer un code"
          icon={TicketIcon}
          size="sm"
          fields={[F.bool("force_nouveau", "Remplacer le code existant")]}
          onSubmit={(b) => genererCodeEnveloppeParent(parentId, b.force_nouveau === true)}
          successMessage="Code généré"
          showResult
          onDone={bump} />
        }>
        <div className="space-y-4">
          <AsyncBlock load={() => getCodeEnveloppeParent(parentId)} deps={[parentId, tick]} isEmpty={(d) => !d}>
            {(c) =>
            <div className="space-y-3">
                {str(c.code) &&
              <div className="rounded-xl border border-dashed border-[var(--em-accent)] bg-[var(--em-accent-soft)] px-4 py-5 text-center">
                    <p className="ecole_m-muted mb-1 text-[12px]">Code à remettre au parent</p>
                    <p className="select-all font-mono text-[24px] font-semibold tracking-[0.25em] text-[var(--em-accent)]">{str(c.code)}</p>
                  </div>
              }
                <ObjectView data={c} hide={["code"]} />
              </div>
            }
          </AsyncBlock>

          <AsyncBlock load={() => listerCodesEnveloppeParent(parentId)} deps={[parentId, tick]} empty={<></>}>
            {(rows) =>
            <div>
                <p className="ecole_m-muted mb-2 text-[12px] font-medium">Historique des codes</p>
                <div className="overflow-hidden rounded-lg border border-[var(--em-border)]">
                  <MiniTable rows={rows} />
                </div>
              </div>
            }
          </AsyncBlock>
        </div>
      </Section>
    </div>);
}
